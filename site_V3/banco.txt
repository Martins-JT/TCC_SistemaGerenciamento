CREATE DATABASE IF NOT EXISTS entrega;

USE ENTREGA;

CREATE TABLE IF NOT EXISTS informacoes
(
ID TINYINT auto_increment PRIMARY KEY,
usuario VARCHAR(40) NOT NULL,
Senha VARCHAR(150) NOT NULL,
endereco VARCHAR(40) NOT NULL,
email VARCHAR(40) NOT NULL,
cep VARCHAR(40) NOT NULL,
produto VARCHAR(50) not null,
caracteristica VARCHAR(50) not null,
foto VARCHAR(100) not null,
permissoes ENUM('usuario', 'admin') not null
);
CREATE TABLE IF NOT EXISTS produtos
(
id_produto TINYINT auto_increment PRIMARY KEY,
usuario VARCHAR(40) NOT NULL,
foto VARCHAR(100) not null,
produto VARCHAR(50) not null,
caracteristica VARCHAR(50) not null,
peso_unidade VARCHAR(50) not null,
estoque VARCHAR(50) not null,
descricao VARCHAR(50) not null,
categoria VARCHAR(50) not null

);

CREATE TABLE IF NOT EXISTS ofertas
(
id_oferta TINYINT auto_increment PRIMARY KEY,
usuario VARCHAR(40) NOT NULL,
foto VARCHAR(100) not null,
produto VARCHAR(50) not null,
caracteristica VARCHAR(50) not null,
peso_unidade VARCHAR(50) not null,
estoque VARCHAR(50) not null,
descricao VARCHAR(50) not null,
categoria VARCHAR(50) not null,
dias VARCHAR(50) not null
);
select * from informacoes;
select * from produtos;
select * from ofertas order by id_oferta asc;
